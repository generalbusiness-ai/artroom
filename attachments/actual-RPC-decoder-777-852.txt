783:   /** The room's newline-delimited JSON bytes, decoded into updates (R-API-8). */
784:   async subscribe(cursor?: Cursor): Promise<UpdateStream> {
785:     const updates = decodeUpdates(await this.#rpc.subscribe(await this.auth(), cursor));
786:     // Each update is looked at for a later activation as it is read (see `actsAt`).
787:     return {
788:       getReader: () => {
789:         const reader = updates.getReader();
790:         return {
791:           read: async () => {
792:             const r = await reader.read();
793:             if (!r.done) this.sawUpdate(r.value);
794:             return r;
795:           },
796:           releaseLock: () => reader.releaseLock(),
797:         };
798:       },
799:       cancel: (reason) => updates.cancel(reason),
800:     };
801:   }
802: }

804: /**
805:  * Decodes a `ByteStream` of UTF-8, newline-delimited JSON `Update`s into an
806:  * `UpdateStream` (R-API-8), with standard streams: the bytes are piped
807:  * through a `TextDecoderStream` and a line-splitting `TransformStream`. A
808:  * line may arrive across several chunks; a chunk may hold several lines.
809:  *
810:  * Ownership: the pipe owns the source; the caller owns the decoded stream.
811:  * `cancel()` ends the subscription whether or not the caller holds a
812:  * reader: a pending read resolves as done, and the cancel reaches the
813:  * source. Every failure, whether a bad byte, a line that is not an update,
814:  * or a failing source, rejects with an `ArtroomError`, never a raw error.
815:  */
816: export function decodeUpdates(bytes: ByteStream): UpdateStream {
817:   const updates = asReadable(bytes).pipeThrough(new TextDecoderStream("utf-8", { fatal: true }) as unknown as ReadableWritablePair<string, Uint8Array>).pipeThrough(splitUpdates());
818:   let active: ReadableStreamDefaultReader<Update> | undefined;
819:   return {
820:     getReader: () => {
821:       const reader = updates.getReader();
822:       active = reader;
823:       return {
824:         read: async () => {
825:           try {
826:             const r = await reader.read();
827:             return r.done ? { done: true as const } : { done: false as const, value: r.value };
828:           } catch (e) {
829:             throw streamError(e);
830:           }
831:         },
832:         releaseLock: () => {
833:           if (active === reader) active = undefined;
834:           reader.releaseLock();
835:         },
836:       };
837:     },
838:     cancel: async (reason?: unknown) => {
839:       try {
840:         if (active !== undefined) {
841:           const reader = active;
842:           active = undefined;
843:           await reader.cancel(reason); // resolves pending reads as done, and cancels back through the pipe
844:           reader.releaseLock();
845:         } else if (!updates.locked) await updates.cancel(reason);
846:       } catch (e) {
847:         throw streamError(e);
848:       }
849:     },
850:   };
851: }
