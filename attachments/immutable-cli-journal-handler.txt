424:  * Runs one act through the journal, whose entry moves from `prepared` to
425:  * `answered` and is removed only after the local steps are done.
426:  *
427:  * - New work: `start` resolves the act from current state; the client hands
428:  *   back the prepared act, which is journaled before it is first sent.
429:  * - `prepared` (sent, no answer kept): a signed act goes straight back to
430:  *   the room with `resubmit`, needing no read session, new signature or
431:  *   current state, so a retired key still gets its receipt (R-IDEM-2). An
432:  *   unsigned bearer act needs its token judged first (R-CRED-10).
433:  * - `answered`: nothing is sent; the local steps run again from the kept
434:  *   answer.
435:  */
436: async function journaled<T>(ctx: Ctx, spec: ActSpec<T>): Promise<{ out: Result<T>; extra: string[]; prepared?: PreparedAct }> {
437:   const { id, room } = roomOf(ctx);
438:   const key = str(ctx.values, "idempotency-key") ?? commandKey();
439:   if (!/^[A-Za-z0-9_-]{1,64}$/.test(key)) throw new UsageError("An idempotency key is 1 to 64 characters from A-Z, a-z, 0-9, '_' and '-'.");
440:   ctx.act = { room: id, key };
441:   let entry: Extract<JournalEntry, { type: "act" }> | undefined = ctx.store.entry(id, "act", key);
442:   if (entry !== undefined && entry.command !== spec.command) {
443:     throw new UsageError(`The idempotency key ${key} belongs to an unfinished artroom ${entry.command}. Repeat that command with it.`);
444:   }
445:   let out: Result<T>;
446:   if (entry?.state === "answered") out = entry.result as Result<T>;
447:   else {
448:     if (entry === undefined) {
449:       const { api } = await open(ctx);
450:       out = await spec.start(api, room, {
451:         idempotencyKey: key,
452:         onPrepared: (prepared) => {
453:           // Fixed now, before anything is sent: what this act owns locally, and the revisions it expects.
454:           const now = ctx.store.read().rooms[id] ?? room;
455:           const local = spec.intent?.(now, prepared, id) ?? { kind: "none" as const };
456:           entry = { v: SCHEMA, type: "act", id: key, room: id, command: spec.command, state: "prepared", prepared, local };
457:           ctx.store.journal(entry);
458:           ctx.step("act-journaled");
459:         },
460:       });
461:       // A preflight refusal or usage error never prepared anything: there is nothing to finish.
462:       if (entry === undefined) {
463:         ctx.act = undefined;
464:         return { out, extra: [] };
465:       }
466:     } else if (entry.prepared.signed !== undefined) {
467:       out = (await resubmit({ url: room.url }, id, entry.prepared.signed, clientOptions(ctx))) as Result<T>;
468:     } else {
469:       const { api } = await open(ctx);
470:       try {
471:         out = (await api.replay(entry.prepared)) as Result<T>;
472:       } catch (e) {
473:         // A bearer act has no signed envelope to keep: once the token is refused, there is nothing to resend (R-CRED-10).
474:         if (isArtroomError(e) && e.code === "unauthenticated") {
475:           ctx.store.finish(id, "act", key);
476:           ctx.act = undefined;
477:           throw { ...e, message: "The bearer token is no longer valid, so this act cannot be sent again. If the room recorded it, a member with a read session can find it with artroom log." };
478:         }
479:         throw e;
480:       }
481:     }
482:     entry = { ...entry!, state: "answered", result: out };
483:     ctx.store.journal(entry);
484:     ctx.step("act-answered");
485:   }
486:   const extra = isRefusal(out) ? [] : applyLocal(ctx, id, key, entry!.local ?? { kind: "none" }, out);
487:   ctx.step("act-completed");
488:   ctx.store.finish(id, "act", key);
489:   ctx.act = undefined;
490:   // The act as it was prepared, and so as it was sent: what the journal kept, whichever run prepared it.
491:   return { out, extra, prepared: entry!.prepared };
492: }
