213:     ...(ctx.values["verbose"] === true ? { log: (l: string) => ctx.io.err(`  ${l}`) } : {}),
214:   };
215: }
216: 
217: type Opened = { api: HttpRoomClient; id: RoomId; room: RoomConfig };
218: 
219: /** Connects once per run, with the full checks; only work that needs a session or new signatures calls it. */
220: function open(ctx: Ctx): Promise<Opened> {
221:   ctx.opened ??= connectRoom(ctx);
222:   return ctx.opened;
223: }
224: 
225: async function connectRoom(ctx: Ctx): Promise<Opened> {
226:   const { id, room } = roomOf(ctx);
227:   let creds;
228:   if (room.custody === "room") {
229:     const token = ctx.store.loadBearer(id);
230:     ctx.secrets.add(token); // registered before any request
231:     creds = { kind: "bearer" as const, token };
232:   } else creds = { kind: "key" as const, signer: await signerFromJwk(ctx.store.loadKey(id).jwk) };
233:   try {
234:     // connect() over HTTPS returns the client's HttpRoomClient, which can also replay a journaled act.
235:     const api = (await connect({ url: room.url }, id, creds, clientOptions(ctx))) as HttpRoomClient;
236:     return { api, id, room };
237:   } catch (e) {
238:     if (isArtroomError(e) && e.code === "unauthenticated" && room.custody === "room") {
239:       // A refused bearer can never resend an unsigned act (R-CRED-10): forget them, and say what to do.
240:       const dropped = ctx.store.forgetBearerActs(id);
241:       throw {
242:         ...e,
243:         message: `The bearer token is no longer valid${dropped > 0 ? `, so ${dropped} unfinished act(s) cannot be sent again` : ""}. Ask an admin for a new MCP invitation. If an act was recorded, a member with a read session can find it with artroom log.`,
244:       };
245:     }
246:     throw e;
247:   }
